## NodeCategory

```typescript
type NodeCategory = 'action' | 'logic' | 'integration' | 'transform';
```

## NodeCapabilities

```typescript
interface NodeCapabilities {
  supportsEnrichment?: boolean;
  supportsBulkActions?: boolean;
  supportsApproval?: boolean;
  supportsRerun?: boolean;
  supportsCancel?: boolean;
}
```

## NodeMetadata

Client-safe metadata (no executor). Used for UI rendering.

<ParamField body="type" type="string" required>
  Unique node type identifier.
</ParamField>
<ParamField body="name" type="string" required>
  Human-readable display name.
</ParamField>
<ParamField body="description" type="string" required>
  Description shown in the editor palette.
</ParamField>
<ParamField body="category" type="NodeCategory" required>
  One of `action`, `logic`, `integration`, `transform`.
</ParamField>
<ParamField body="estimatedDuration" type="number">
  Estimated execution time in seconds.
</ParamField>
<ParamField body="capabilities" type="NodeCapabilities">
  Feature flags for the node.
</ParamField>

## NodeDefinition\<TInput, TOutput\>

Complete node definition. Extends `NodeMetadata` with schemas and executor.

<ParamField body="inputSchema" type="z.ZodSchema<TInput>" required>
  Zod schema for validating node inputs.
</ParamField>
<ParamField body="outputSchema" type="z.ZodSchema<TOutput>" required>
  Zod schema for validating node outputs.
</ParamField>
<ParamField body="executor" type="NodeExecutor<TInput, TOutput>" required>
  Async function that executes the node logic.
</ParamField>

## NodeExecutionContext

Passed to every node executor.

<ParamField body="userId" type="string" required>
  The user running the workflow.
</ParamField>
<ParamField body="campaignId" type="string">
  Optional project/campaign context.
</ParamField>
<ParamField body="workflowExecutionId" type="string" required>
  Unique identifier for this workflow run.
</ParamField>
<ParamField body="variables" type="Record<string, unknown>" required>
  All workflow variables (includes upstream node outputs).
</ParamField>
<ParamField body="resolveNestedPath" type="(path: string) => unknown" required>
  Resolve dot-notation paths like `contact.email` against variables.
</ParamField>
<ParamField body="services" type="NodeServices">
  Injected service implementations.
</ParamField>

## NodeExecutionResult\<TOutput\>

Returned by every node executor.

<ParamField body="success" type="boolean" required>
  Whether execution succeeded.
</ParamField>
<ParamField body="output" type="TOutput">
  The node's output data (must match `outputSchema`).
</ParamField>
<ParamField body="error" type="string">
  Error message if `success` is `false`.
</ParamField>
<ParamField body="nextNodeId" type="string">
  For conditional branching — which node to execute next.
</ParamField>
<ParamField body="needsApproval" type="NodeApprovalRequest">
  Request human approval before continuing.
</ParamField>
<ParamField body="notification" type="object">
  Emit a notification with `title`, `message`, and optional `data`.
</ParamField>

## NodeExecutor\<TInput, TOutput\>

```typescript
type NodeExecutor<TInput = unknown, TOutput = unknown> = (
  input: TInput,
  context: NodeExecutionContext
) => Promise<NodeExecutionResult<TOutput>>;
```

## NodeApprovalRequest

```typescript
interface NodeApprovalRequest {
  resourceIds: string[];
  resourceType: string;
  message?: string;
  detailComponent?: string;
}
```

## BaseNodeConfig / Approval / Notifications

Optional config any node can include:

```typescript
interface BaseNodeConfig {
  approval?: NodeApprovalConfig;
  notification?: NodeNotificationConfig;
}

interface NodeApprovalConfig {
  required: boolean;
  pauseWorkflow?: boolean;       // default: true
  timeoutMinutes?: number;       // default: 1440 (24h)
  approvalType?: string;
  message?: string;
}

interface NodeNotificationConfig {
  enabled: boolean;
  channels?: Array<'chat' | 'email' | 'slack' | 'webhook'>;
  message?: string;
  priority?: 'low' | 'medium' | 'high';
  notifyOnComplete?: boolean;
  notifyOnError?: boolean;
}
```

## NodeServices

All optional. Host applications inject implementations via `context.services`.

```typescript
interface NodeServices {
  apollo?: ApolloClient;
  dataForSeo?: DataForSeoClient;
  twitter?: TwitterClient;
  forumScout?: ForumScoutClient;
  openai?: OpenAIClient;
  anthropic?: AnthropicClient;
  notifications?: NotificationService;
  storage?: StorageService;
  cache?: CacheService;
  emailDrafts?: EmailDraftsService;
  analyzedPosts?: AnalyzedPostsStorage;
}
```

### Service Interfaces

| Service | Key Methods |
|---------|------------|
| `ApolloClient` | `searchContacts(params)`, `enrichContact(contactId)` |
| `DataForSeoClient` | `getOnPageInstant(url, options?)`, `getRelatedKeywords(keywords, options?)` |
| `TwitterClient` | `searchTweets(query, options?)` |
| `ForumScoutClient` | `searchLinkedIn(keywords, options?)` |
| `OpenAIClient` | `generateVideo(params)`, `generateImage(params)` |
| `AnthropicClient` | `generateText(params)`, `generateStructured<T>(params)` |
| `NotificationService` | `send(params)` |
| `EmailDraftsService` | `createDraft(params)` |
| `AnalyzedPostsStorage` | `storePosts(params)` |
| `StorageService` | `save<T>(key, data)`, `get<T>(key)`, `delete(key)` |
| `CacheService` | `get<T>(key)`, `set<T>(key, value, ttlSeconds?)`, `delete(key)` |
